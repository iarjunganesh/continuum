# Floors are the versions this project is actually tested against; caps stop a
# major release from landing in CI unannounced. CI installs from this file on a
# clean runner (.github/workflows/ci.yml), so an unbounded floor means the next
# upstream major becomes a surprise CI failure — or worse, a surprise *pass*
# against changed semantics. Bump deliberately, run `make test`, then commit.
#
# Requires Python 3.14 (see pyproject.toml).

# Core
fastapi>=0.141,<1
uvicorn[standard]>=0.52,<1
# CEILING IS A COMPAT PIN, NOT THE USUAL MAJOR CAP. Hugging Face's Space builder
# installs `gradio[oauth,mcp]==<sdk_version>` on top of this file, and gradio's
# `mcp` extra pins `pydantic<=2.12.5,>=2.11.10` (true of every gradio 6.x, so
# moving sdk_version does not help). A floor of >=2.13 makes the Space build
# unresolvable — which is how v0.7.0 broke it. The floor is set by `mcp` 1.29,
# which needs pydantic>=2.12.0 on Python 3.14. Capping here rather than leaving
# it open keeps CI resolving the same pydantic the Space runs. Revisit when
# gradio's mcp extra lifts its cap.
pydantic>=2.12,<2.12.6
pydantic-settings>=2.14,<3

# Database — psycopg 3 (psycopg2 has no Python 3.14 wheels)
psycopg[binary]>=3.3,<4

# AWS
boto3>=1.43,<2

# CockroachDB Cloud Managed MCP Server client (agents/query_agent.py).
# PINNED BELOW 2.0 DELIBERATELY. mcp 2.0.0 is a major release and query_agent
# talks to a *managed* server whose contract we don't control; the unit tests
# mock the SDK at the import boundary (tests/unit/test_query_agent.py), so they
# would pass against a 2.x client that fails against the real server. Lifting
# this cap requires a live round trip against the Managed MCP Server, not a
# green unit suite. See ADR 003.
mcp>=1.28,<2

# HTTP / demo driver
httpx>=0.28,<1

# Cross-platform process control (chaos_kill.py)
psutil>=7.2,<8

# Observability
structlog>=26.1,<27

# Demo UI. MUST match `sdk_version` in the README frontmatter — that's what the
# Hugging Face Space builds against, and a mismatch fails at Space build time,
# not here.
gradio>=6.22,<7

# Dev / test
pytest>=9.1,<10
pytest-cov>=7.1,<8
ruff>=0.16,<0.17
